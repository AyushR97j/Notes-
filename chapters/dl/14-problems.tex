\chapter{Problems}
\label{ch:problems}
\bigproblemset

Seventy-three problems in the shape of online assessments and technical rounds: one-minute
numerics, chains of shapes and counts, ``which statements are true'', and short derivations.
Every number in the hints and solutions is computed (and where possible cross-checked against
PyTorch) by \script{c14_problems}. Time yourself: a chili-free OA item should take under two
minutes.

\begin{problem}\onechili

\begin{problem}\twochili
	Explain in three sentences why the Transformer replaced the RNN for machine translation.
	\begin{hint}Parallelism, path length, scaling.\end{hint}
	\begin{sol}Self-attention processes all positions of a sentence in parallel during training, while an RNN must step through time. Any two
	positions interact through one attention operation instead of a chain of recurrent steps, so long-range dependencies are easier to learn
	and gradients do not vanish across the sentence. Together these let much larger models be trained on much more data, and quality kept
	improving with scale.\end{sol}
\end{problem}

\begin{problem}\twochili
	Show that the BatchNorm output is unchanged if every input in the batch is multiplied by $c>0$ and shifted by $d$, and conclude what this
	means for the gradient with respect to the inputs.
	\begin{hint}$\mu\to c\mu+d$, $\sigma\to c\sigma$ (ignoring $\eps$).\end{hint}
	\begin{sol}$(cx_i+d-(c\mu+d))/(c\sigma)=(x_i-\mu)/\sigma$. Because the output is invariant along the directions $\one$ (shift) and $\x$
	(scale), the input gradient is orthogonal to both: it sums to zero and has zero inner product with $\hat\x$.\end{sol}
\end{problem}

\begin{problem}\twochili
	Show that the softmax Jacobian is $\diag(\bm p)-\bm p\bm p\T$ and that its rows sum to zero.
	\begin{hint}Differentiate $p_i=e^{z_i}/\sum_ke^{z_k}$.\end{hint}
	\begin{sol}$\partial p_i/\partial z_j=p_i\delta_{ij}-e^{z_i}e^{z_j}/(\sum_ke^{z_k})^2=p_i(\delta_{ij}-p_j)$. Row sum:
	$p_i(1-\sum_jp_j)=0$, because the probabilities always sum to one.\end{sol}
\end{problem}

\begin{problem}\threechili
	Prove that two stacked $3\times3$ convolutions (stride 1, no non-linearity between them) are equivalent to one $5\times5$ convolution, and
	explain why the stacked version is still preferred.
	\begin{hint}Convolution is associative: $K_2\star(K_1\star X)=(K_2*K_1)\star X$.\end{hint}
	\begin{sol}Each output of the second layer is a linear combination of a $3\times3$ window of first-layer outputs, each of which is a linear
	combination of a $3\times3$ input window; the union of windows is $5\times5$ and the composite weights form a $5\times5$ kernel (the
	convolution of the two kernels). With a ReLU in between this equivalence breaks, which is the point: the stack has more non-linearity and,
	per channel pair, $18$ instead of $25$ weights, though it can only express rank-constrained $5\times5$ kernels.\end{sol}
\end{problem}

\begin{problem}\threechili
	Show that adding a constant $c$ to every logit leaves cross-entropy unchanged, and that scaling every logit by $a>1$ lowers the loss of a
	correct (arg-max) prediction. What does this imply for training with cross-entropy on separable data?
	\begin{hint}$\CE=\LSE(\z)-z_y$.\end{hint}
	\begin{sol}$\LSE(\z+c)-(z_y+c)=\LSE(\z)-z_y$. If $z_y$ is the strict maximum, $\LSE(a\z)-az_y=\log\sum_ke^{a(z_k-z_y)}$ decreases in $a$
	because every exponent with $k\ne y$ is negative and becomes more negative. On separable data the loss can therefore always be reduced by
	scaling the weights up: without regularisation the weights grow without bound, which is one reason for weight decay and label smoothing.\end{sol}
\end{problem}
