\chapter{Problems}
\label{ch:problems}
\bigproblemset
Sixty-seven problems in the shapes that online assessments use: one-minute numericals, ``which statements are
true'' multi-selects, and short derivations. They are original; every number in the solutions is produced by
\script{c19_problems}. Hints and full solutions are in the appendices (each problem links to both). Aim for a
minute or two per numerical with only a basic calculator.

\section{Regression and regularisation}
\prototype{Fit a line to five points in under two minutes.}

\begin{problem}[Least-squares line]\onechili
	Fit $y=b_0+b_1x$ to $(2,3),(4,5),(6,6),(8,9),(10,12)$. Give $b_1$, $b_0$, the prediction at $x=7$ and $R^2$.
	\begin{hint}$\bar x=6$, $\bar y=7$; $b_1=\sum(x-\bar x)(y-\bar y)/\sum(x-\bar x)^2$.\end{hint}
	\begin{sol}$\sum(x-\bar x)(y-\bar y)=\gv{c19_problems/p1_sxy}$, $\sum(x-\bar x)^2=\gv{c19_problems/p1_sxx}$, so
		$b_1=\gv{c19_problems/p1_b1}$, $b_0=7-1.1\cdot6=\gv{c19_problems/p1_b0}$. At $x=7$: $\gv{c19_problems/p1_pred}$.
		$R^2=\gv{c19_problems/p1_r2}$.\end{sol}
\end{problem}

\begin{problem}[Correlation from a slope]\onechili
	A regression of $y$ on $x$ has slope $1.5$; $s_x=4$, $s_y=10$. Find $r$ and the fraction of variance explained.
	\begin{hint}$b_1=r\,s_y/s_x$.\end{hint}
	\begin{sol}$r=1.5\cdot4/10=\gv{c19_problems/p2_r}$; $R^2=r^2=\gv{c19_problems/p2_r2}$.\end{sol}
\end{problem}

\begin{problem}[Scoring a line]\onechili

	\begin{sol}Every residual has magnitude $0.5$: MSE $=\gv{c19_problems/p3_mse}$, MAE $=\gv{c19_problems/p3_mae}$.\end{sol}
\end{problem}

\begin{problem}[One-feature ridge]\onechili
	Centred data with $\sum x_iy_i=12$ and $\sum x_i^2=8$. Give the OLS slope and the ridge slope for $\lambda=4$.
	\begin{hint}$\sum xy/(\sum x^2+\lambda)$.\end{hint}
	\begin{sol}OLS $=\gv{c19_problems/p4_ols}$; ridge $=12/12=\gv{c19_problems/p4_ridge}$.\end{sol}
\end{problem}

\begin{problem}[Soft thresholding]\onechili
	With an orthonormal design the OLS coefficients are $(1.8,-0.3,0.6)$. Give the lasso solution with threshold
	$\lambda=0.5$ and the ridge solution $\hat w/(1+\lambda)$ with $\lambda=0.5$.
	\begin{hint}$\sign(z)\max(\abs z-\lambda,0)$.\end{hint}
	\begin{sol}Lasso $(\gv{c19_problems/p5_out})$; ridge $(\gv{c19_problems/p5_ridge})$. Only lasso zeroes a
		coefficient.\end{sol}
\end{problem}

\begin{problem}[Adjusted $R^2$]\onechili
	$n=30$. Model A: 2 features, $R^2=0.62$. Model B: 6 features, $R^2=0.66$. Compare adjusted $R^2$.
	\begin{hint}$1-(1-R^2)(n-1)/(n-p-1)$.\end{hint}
	\begin{sol}A: $\gv{c19_problems/p6_a}$; B: $\gv{c19_problems/p6_b}$. A is preferred: four extra features bought
		too little.\end{sol}
\end{problem}

\begin{problem}[The augmented error]\onechili
	For $E_{\text{aug}}(\w)=E_{\text{in}}(\w)+\lambda\norm{\w}^2$, which are true? (a) $\lambda=0$ gives the
	unregularised fit. (b) Large $\lambda$ lowers training error. (c) Large $\lambda$ increases bias. (d) $\lambda$ should
	be chosen to minimise $E_{\text{in}}$. (e) As $\lambda\to\infty$ the weights go to zero.
	\begin{hint}Two are false.\end{hint}
	\begin{sol}True: (a), (c), (e). False: (b) --- training error can only rise; (d) --- minimising $E_{\text{in}}$
		always picks $\lambda=0$; use validation.\end{sol}
\end{problem}

\begin{problem}[Sigmoid derivative and log-loss gradient]\twochili
	Show $\sigma'(z)=\sigma(z)(1-\sigma(z))$ and hence $\partial\ell/\partial z=\sigma(z)-y$ for the log-loss.
	\begin{hint}Differentiate $(1+e^{-z})^{-1}$.\end{hint}
	\begin{sol}$\sigma'=e^{-z}/(1+e^{-z})^2=\sigma\cdot(1-\sigma)$. For $\ell=-y\log\sigma-(1-y)\log(1-\sigma)$:
		$\ell'=-y(1-\sigma)+(1-y)\sigma=\sigma-y$.\end{sol}
\end{problem}

\begin{problem}[The 63.2 percent]\twochili
	Show that $1-(1-1/n)^n\to1-e^{-1}$ and evaluate it at $n=100$.
	\begin{hint}Take logs and use $\log(1-u)=-u+O(u^2)$.\end{hint}
	\begin{sol}$n\log(1-1/n)=n(-1/n-1/(2n^2)-\dots)\to-1$, so $(1-1/n)^n\to e^{-1}$. At $n=100$:
		$\gv{c19_problems/p63_100}$.\end{sol}
\end{problem}

\begin{problem}[Ridge closed form]\twochili
	Derive the minimiser of $\norm{\y-\X\w}^2+\lambda\norm{\w}^2$ and show it is unique for $\lambda>0$.
	\begin{hint}Set the gradient to zero; check the Hessian.\end{hint}
	\begin{sol}Gradient $-2\X\T(\y-\X\w)+2\lambda\w=0$ gives $(\X\T\X+\lambda\I)\w=\X\T\y$. The Hessian
		$2(\X\T\X+\lambda\I)$ has eigenvalues at least $2\lambda>0$, so the objective is strictly convex and the
		minimiser unique.\end{sol}
\end{problem}

\begin{problem}[Why $k$-means terminates]\twochili
	Prove that Lloyd's algorithm stops after finitely many iterations.
	\begin{hint}Monotone decrease plus finitely many partitions.\end{hint}
	\begin{sol}The assignment step minimises inertia over assignments for fixed centroids; the update step minimises it
		over centroids for fixed assignments (the mean minimises a sum of squared distances). So inertia never increases.
		There are finitely many partitions of $n$ points into $k$ clusters, and a partition determines the centroids, so
		unless the partition repeats --- at which point nothing changes and the algorithm stops --- inertia strictly
		decreases through distinct partitions, which cannot go on forever.\end{sol}
\end{problem}

\begin{problem}[AUC as a probability]\threechili
	Explain why the area under the empirical ROC curve equals the fraction of positive--negative pairs ranked correctly
	(no ties).
	\begin{hint}Sweep the threshold down through the sorted scores.\end{hint}
	\begin{sol}Sorting by score and lowering the threshold, each positive moves the curve up by $1/P$ and each negative
		right by $1/N$. A negative step adds a column of area equal to (current TPR)$/N$, where TPR counts positives scored
		above that negative. Summing over negatives gives $\frac1{PN}\sum_{\text{neg}}\#\{\text{pos above it}\}$, the
		fraction of correctly ordered pairs.\end{sol}
\end{problem}

\begin{problem}[MAP shrinkage]\twochili
	With prior $\mathrm{Beta}(a,b)$ and $h$ successes in $n$ trials, show the MAP estimate lies between the MLE $h/n$ and
	the prior mode, and moves towards the MLE as $n$ grows.
	\begin{hint}Write the MAP as a weighted average.\end{hint}
	\begin{sol}$\hat p=\frac{h+a-1}{n+a+b-2}=\frac{n}{n+a+b-2}\cdot\frac hn+\frac{a+b-2}{n+a+b-2}\cdot\frac{a-1}{a+b-2}$:
		a convex combination of the MLE and the prior mode with weight $n/(n+a+b-2)\to1$ on the MLE.\end{sol}
\end{problem}
